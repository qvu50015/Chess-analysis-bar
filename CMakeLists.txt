cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS version")
endif()
project(chess_eval_bar VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON CACHE BOOL "Export compiler commands for editor tooling")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT UNIX)
  message(FATAL_ERROR "M1 subprocess support requires macOS or another POSIX system")
endif()
add_library(chess_eval src/engine/uci.cpp src/engine/process_posix.cpp src/engine/engine.cpp src/chess/fen.cpp)
target_include_directories(chess_eval PUBLIC include)
target_compile_options(chess_eval PRIVATE -Wall -Wextra -Wpedantic)
add_executable(chess-eval src/main.cpp)
target_link_libraries(chess-eval PRIVATE chess_eval)
target_compile_options(chess-eval PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit_tests.cpp)
  target_link_libraries(unit_tests PRIVATE chess_eval)
  add_test(NAME unit COMMAND unit_tests)
  add_executable(fake_engine tests/fake_engine.cpp)
  add_executable(engine_tests tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE chess_eval)
  add_test(NAME engine_lifecycle COMMAND engine_tests $<TARGET_FILE:fake_engine>)
  set_tests_properties(engine_lifecycle PROPERTIES TIMEOUT 15)
  set(STOCKFISH_EXECUTABLE "" CACHE FILEPATH "Optional Stockfish executable for integration tests")
  if(STOCKFISH_EXECUTABLE)
    add_test(NAME stockfish_integration COMMAND chess-eval --stockfish "${STOCKFISH_EXECUTABLE}" --fen "rnbqkbnr/pppppppp/8/8/8/8/PPPPPPPP/RNBQKBNR w KQkq - 0 1" --depth 6)
    set_tests_properties(stockfish_integration PROPERTIES PASS_REGULAR_EXPRESSION "cp -?[0-9]+" TIMEOUT 35)
  endif()
endif()

option(BUILD_GUI "Build the Qt 6 evaluation bar" ON)
if(BUILD_GUI)
  find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
  add_library(chess_capture
    src/capture/types.cpp src/capture/backend.cpp src/capture/manager.cpp
    include/chess_eval/capture/manager.hpp)
  set_target_properties(chess_capture PROPERTIES AUTOMOC ON)
  target_include_directories(chess_capture PUBLIC include)
  target_link_libraries(chess_capture PUBLIC Qt6::Widgets)
  target_compile_options(chess_capture PRIVATE -Wall -Wextra -Wpedantic)
  if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    target_sources(chess_capture PRIVATE src/capture/mac_backend.mm)
    set_source_files_properties(src/capture/mac_backend.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(chess_capture PRIVATE
      "-framework ScreenCaptureKit" "-framework CoreGraphics"
      "-framework CoreMedia" "-framework CoreVideo" "-framework Foundation")
  else()
    target_sources(chess_capture PRIVATE src/capture/unsupported_backend.cpp)
  endif()
  add_library(chess_eval_ui
    src/ui/controller.cpp src/ui/evaluation_bar.cpp src/ui/window.cpp
    src/ui/capture_panel.cpp src/ui/region_selector.cpp include/chess_eval/ui/capture_panel.hpp
    include/chess_eval/ui/controller.hpp)
  set_target_properties(chess_eval_ui PROPERTIES AUTOMOC ON)
  target_link_libraries(chess_eval_ui PUBLIC chess_eval chess_capture Qt6::Widgets)
  target_compile_options(chess_eval_ui PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(chess-eval-bar src/ui/main.cpp)
  target_link_libraries(chess-eval-bar PRIVATE chess_eval_ui)
  if(BUILD_TESTING)
    if(APPLE)
      add_executable(native_capture_tests tests/native_capture_tests.mm)
      target_compile_options(native_capture_tests PRIVATE -fobjc-arc -Wall -Wextra -Wpedantic)
      target_link_libraries(native_capture_tests PRIVATE chess_capture)
      add_test(NAME native_capture_lifetime COMMAND native_capture_tests)
      set_tests_properties(native_capture_lifetime PROPERTIES TIMEOUT 10)
    endif()
    add_executable(capture_tests tests/capture_tests.cpp)
    target_link_libraries(capture_tests PRIVATE chess_eval_ui)
    add_test(NAME capture_logic COMMAND capture_tests)
    set_tests_properties(capture_logic PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE chess_eval_ui)
    add_test(NAME qt_evaluation COMMAND ui_tests $<TARGET_FILE:fake_engine>)
    set_tests_properties(qt_evaluation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  endif()
endif()
